\begin{figure}[htbp]
\centering
\begin{tikzpicture}[
    x=1cm,y=1cm,
    line cap=round,
    line join=round,
    ray/.style={line width=0.9pt},
    boundary/.style={line width=1.25pt},
    construction/.style={line width=0.45pt,dashed},
    dimension/.style={line width=0.45pt,{Latex[length=2.2mm]}-{Latex[length=2.2mm]}},
    maparrow/.style={line width=0.8pt,-{Latex[length=3mm]}},
    point/.style={circle,fill=black,inner sep=1.6pt},
    every node/.style={font=\small}
]
% --------------------------------------------------------------------------
% (a) Physical coordinates
% --------------------------------------------------------------------------
\begin{scope}[shift={(0,0)}]
  \node[font=\normalsize] at (2.75,3.95) {(a) Physical coordinates $(x,h)$};

  % Opaque planar boundary
  \draw[boundary] (0.15,0) -- (5.35,0);
  \node[anchor=north west] at (0.22,-0.12) {opaque boundary $h=0$};

  % Exact exponential grazing branch, normalized with R=1.
  \draw[ray,domain=-1.02:0,samples=100,smooth,variable=\u]
    plot ({4.55+3.65*\u},{-2.35*ln(cos(57.2958*\u))});

  % Tangency and endpoint
  \coordinate (G) at (4.55,0);
  \coordinate (O) at ({4.55-3.65*1.02},{-2.35*ln(cos(57.2958*1.02))});
  \node[point] at (G) {};
  \node[point] at (O) {};
  \node[below=2pt] at (G) {$G$};
  \node[above left=1pt] at (O) {$O$};

  % Coordinate guides
  \draw[construction] (O) -- ($(O |- G)$);
  \draw[dimension] ($(O |- G)+(-0.22,0)$) -- node[left] {$h$} ($(O)+(-0.22,0)$);
  \draw[->,line width=0.45pt] (0.28,0.36) -- (1.08,0.36) node[right] {$x$};
  \draw[->,line width=0.45pt] (0.28,0.36) -- (0.28,1.16) node[above] {$h$};

  % Tangent marker and ray label
  \draw[line width=0.45pt] ($(G)+(-0.42,0.12)$) -- ($(G)+(0.42,0.12)$);
  \node at (3.05,2.55) {curved grazing ray};
  \draw[-{Latex[length=2.3mm]},line width=0.55pt]
    (2.80,2.35) -- (2.47,1.92);
\end{scope}

% --------------------------------------------------------------------------
% Mapping arrow
% --------------------------------------------------------------------------
\draw[maparrow] (5.75,1.95) -- (7.05,1.95);
\node[align=center] at (6.40,2.72)
  {$r=R e^{h/R}$\\[-1pt]$\phi=x/R$};

% --------------------------------------------------------------------------
% (b) Conformal coordinates
% --------------------------------------------------------------------------
\begin{scope}[shift={(7.25,0)}]
  \node[font=\normalsize] at (2.65,3.95) {(b) Conformal coordinates $(r,\phi)$};

  \coordinate (C) at (2.55,1.62);
  \def\R{1.35}
  \coordinate (Gt) at ($(C)+(0,\R)$);
  \coordinate (Ot) at ($(Gt)+(-2.15,0)$);

  % Circular boundary and transformed ray
  \draw[boundary] (C) circle[radius=\R];
  \draw[ray] ($(Ot)+(-0.18,0)$) -- ($(Gt)+(1.15,0)$);
  \node[point] at (Gt) {};
  \node[point] at (Ot) {};
  \node[above right=1pt] at (Gt) {$G'$};
  \node[above left=1pt] at (Ot) {$O'$};
  \node[below=3pt] at ($(C)+(0,-\R)$) {circular boundary $r=R$};
  \node[above] at ($(Gt)+(-0.82,0.10)$) {straight ray};

  % Radius to tangency point and to endpoint
  \draw[line width=0.55pt] (C) -- (Gt) node[midway,right] {$R$};
  \draw[line width=0.55pt] (C) -- (Ot) node[pos=0.57,below left] {$r$};
  \node[point,inner sep=1.2pt] at (C) {};
  \node[below right=1pt] at (C) {$C$};

  % Right-angle marker at tangency
  \draw[line width=0.45pt]
    ($(Gt)+(-0.18,0)$) -- ($(Gt)+(-0.18,-0.18)$) -- ($(Gt)+(0,-0.18)$);

  % Angular displacement arc
  \draw[-{Latex[length=2mm]},line width=0.5pt]
    ($(C)+(90:0.62)$) arc[start angle=90,end angle=132.5,radius=0.62];
  \node at ($(C)+(111:0.92)$) {$\Delta\phi$};

  % Coordinate orientation
  \draw[->,line width=0.45pt] ($(C)+(2.05,-0.95)$) -- ++(0.72,0)
    node[right] {$r$};
  \draw[-{Latex[length=2mm]},line width=0.45pt]
    ($(C)+(2.12,-0.38)$) arc[start angle=-35,end angle=35,radius=0.52];
  \node at ($(C)+(2.83,-0.05)$) {$\phi$};
\end{scope}
\end{tikzpicture}
\caption{Conformal representation of the exponential refractive model. (a) In physical coordinates, a grazing ray curves above the opaque planar boundary and is tangent to it at $G$. (b) Under $r=R e^{h/R}$ and $\phi=x/R$, the boundary becomes the circle $r=R$, while the same ray becomes a straight tangent line at $G'$. Physical height maps nonlinearly to radius, and horizontal displacement maps to angle.}
\label{fig:conformal-mapping}
\end{figure}
