]
% --------------------------------------------------------------------------
% (a) Physical plane
% --------------------------------------------------------------------------
\begin{scope}[shift={(0,0)}]
  \node[font=\normalsize] at (2.35,3.80) {(a) Physical plane};

  \draw[boundary] (0.10,0) -- (4.65,0);
  \node[anchor=north west] at (0.18,-0.12) {$h=0$};

  % Exact grazing branch, shown in normalized form and affinely scaled.
  \draw[ray,domain=-1.02:0,samples=120,smooth,variable=\u]
    plot ({4.03+3.28*\u},{-2.14*ln(cos(57.2958*\u))});

  \coordinate (G) at (4.03,0);
  \coordinate (O) at ({4.03-3.28*1.02},{-2.14*ln(cos(57.2958*1.02))});

  \node[point] at (G) {};
  \node[point] at (O) {};
  \node[below=2pt] at (G) {$G$};
  \node[above left=1pt] at (O) {$O$};

  \draw[construction] (O) -- ($(O |- G)$);

  % Compact coordinate axes.
  \coordinate (A) at (0.42,0.38);
  \draw[->,line width=0.45pt] (A) -- ++(0.72,0) node[right] {$x$};
  \draw[->,line width=0.45pt] (A) -- ++(0,0.72) node[above] {$h$};

  % Tangent marker.
  \draw[line width=0.45pt] ($(G)+(-0.34,0.10)$) -- ($(G)+(0.34,0.10)$);
\end{scope}

% --------------------------------------------------------------------------
% Conformal map
% --------------------------------------------------------------------------
\begin{scope}[shift={(4.62,0)}]
  \node[align=center] at (1.20,2.98)
    {$r=R e^{h/R}$\\[-1pt]$\phi=x/R$};
  \draw[maparrow] (0.10,1.76) -- (2.30,1.76);
\end{scope}

% --------------------------------------------------------------------------
% (b) Conformal image
% --------------------------------------------------------------------------
\begin{scope}[shift={(7.12,0)}]
  \node[font=\normalsize] at (2.18,3.80) {(b) Conformal image};

  \coordinate (C) at (2.15,1.47);
  \def\R{1.06}
  \coordinate (Gp) at ($(C)+(0,\R)$);
  \coordinate (Op) at ($(Gp)+(-2.12,0)$);

  \draw[boundary] (C) circle[radius=\R];
  \draw[tangent] ($(Op)+(-0.08,0)$) -- ($(Gp)+(1.02,0)$);

  \node[point] at (Gp) {};
  \node[point] at (Op) {};
  \node[point,inner sep=1.2pt] at (C) {};

  \node[above right=1pt] at (Gp) {$G'$};
  \node[above left=1pt] at (Op) {$O'$};
  \node[below right=1pt] at (C) {$C$};

  \draw[line width=0.55pt] (C) -- (Gp) node[midway,right] {$R$};
  \draw[line width=0.55pt] (C) -- (Op) node[pos=0.53,below left] {$r$};

  % Right-angle marker at tangency.
  \draw[line width=0.45pt]
    ($(Gp)+(-0.18,0)$) -- ($(Gp)+(-0.18,-0.18)$) -- ($(Gp)+(0,-0.18)$);

  % Angular displacement.
  \draw[-{Latex[length=2mm]},line width=0.5pt]
    ($(C)+(90:0.52)$) arc[start angle=90,end angle=136.8,radius=0.52];
  \node at ($(C)+(113:0.78)$) {$\Delta\phi$};

  \node[below=5pt,align=center] at ($(C)+(0,-\R)$)
    {circular boundary\\[-1pt]$r=R$};
\end{scope}
