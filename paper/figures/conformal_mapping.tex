\begin{tikzpicture}[
    x=1cm,y=1cm,
    line cap=round,
    line join=round,
    ray/.style={line width=0.95pt},
    boundary/.style={line width=1.35pt},
    construction/.style={line width=0.45pt,dashed},
    maparrow/.style={line width=0.9pt,-{Latex[length=3mm]}},
    point/.style={circle,fill=black,inner sep=1.7pt},
    every node/.style={font=\small}
]
% --------------------------------------------------------------------------
% (a) Physical plane
% --------------------------------------------------------------------------
\begin{scope}[shift={(0,0)}]
  \node[font=\normalsize] at (2.45,3.85) {(a) Physical plane};

  \draw[boundary] (0.10,0) -- (4.80,0);
  \node[anchor=north west] at (0.18,-0.12) {$h=0$};

  % Exact grazing branch, shown in normalized form and affinely scaled.
  \draw[ray,domain=-1.02:0,samples=120,smooth,variable=\u]
    plot ({4.12+3.35*\u},{-2.18*ln(cos(57.2958*\u))});

  \coordinate (G) at (4.12,0);
  \coordinate (O) at ({4.12-3.35*1.02},{-2.18*ln(cos(57.2958*1.02))});

  \node[point] at (G) {};
  \node[point] at (O) {};
  \node[below=2pt] at (G) {$G$};
  \node[above left=1pt] at (O) {$O$};

  \draw[construction] (O) -- ($(O |- G)$);

  % Compact coordinate axes.
  \coordinate (A) at (0.42,0.38);
  \draw[->,line width=0.45pt] (A) -- ++(0.72,0) node[right] {$x$};
  \draw[->,line width=0.45pt] (A) -- ++(0,0.72) node[above] {$h$};

  % Tangent marker.
  \draw[line width=0.45pt] ($(G)+(-0.34,0.10)$) -- ($(G)+(0.34,0.10)$);
\end{scope}

% --------------------------------------------------------------------------
% Conformal map
% --------------------------------------------------------------------------
\begin{scope}[shift={(5.05,0)}]
  \node[align=center] at (0.95,2.75)
    {$r=R e^{h/R}$\\[-1pt]$\phi=x/R$};
  \draw[maparrow] (0.20,1.75) -- (1.70,1.75);
\end{scope}

% --------------------------------------------------------------------------
% (b) Conformal image
% --------------------------------------------------------------------------
\begin{scope}[shift={(7.05,0)}]
  \node[font=\normalsize] at (2.35,3.85) {(b) Conformal image};

  \coordinate (C) at (2.25,1.46);
  \def\R{1.18}
  \coordinate (Gp) at ($(C)+(0,\R)$);
  \coordinate (Op) at ($(Gp)+(-1.92,0)$);

  \draw[boundary] (C) circle[radius=\R];
  \draw[ray] ($(Op)+(-0.12,0)$) -- ($(Gp)+(0.98,0)$);

  \node[point] at (Gp) {};
  \node[point] at (Op) {};
  \node[point,inner sep=1.2pt] at (C) {};

  \node[above right=1pt] at (Gp) {$G'$};
  \node[above left=1pt] at (Op) {$O'$};
  \node[below right=1pt] at (C) {$C$};

  \draw[line width=0.55pt] (C) -- (Gp) node[midway,right] {$R$};
  \draw[line width=0.55pt] (C) -- (Op) node[pos=0.55,below left] {$r$};

  % Right-angle marker at tangency.
  \draw[line width=0.45pt]
    ($(Gp)+(-0.18,0)$) -- ($(Gp)+(-0.18,-0.18)$) -- ($(Gp)+(0,-0.18)$);

  % Angular displacement.
  \draw[-{Latex[length=2mm]},line width=0.5pt]
    ($(C)+(90:0.56)$) arc[start angle=90,end angle=132.5,radius=0.56];
  \node at ($(C)+(111:0.84)$) {$\Delta\phi$};

  \node[below=5pt] at ($(C)+(0,-\R)$) {$r=R$};
\end{scope}
\end{tikzpicture}
\caption{Conformal representation of the exponential refractive model. The transformation $r=R e^{h/R}$, $\phi=x/R$ maps the opaque planar boundary $h=0$ to the circle $r=R$ and the curved grazing ray to a straight tangent line.}
\label{fig:conformal-mapping}
